<script>
// 本地存储工具
const storage = {
    get(){return JSON.parse(localStorage.getItem('errorBook')||'[]')},
    save(arr){localStorage.setItem('errorBook',JSON.stringify(arr))}
}
let currentVideoBlob = null;

// 图片预览
const imgFile = document.getElementById('imgFile');
const imgPreview = document.getElementById('imgPreview');
imgFile.onchange = async e=>{
    const file = e.target.files[0];
    if(!file)return;
    const url = URL.createObjectURL(file);
    imgPreview.src = url;
    imgPreview.classList.remove('hidden');
}

// OCR识别
document.getElementById('ocrBtn').onclick = async ()=>{
    const file = imgFile.files[0];
    if(!file){alert("请先上传题目图片");return;}
    const ocrBtn = document.getElementById('ocrBtn');
    ocrBtn.innerText="识别中...";
    ocrBtn.disabled=true;
    const result = await Tesseract.recognize(file,'chi_sim+eng',{logger:m=>console.log(m)});
    document.getElementById('questionText').value = result.data.text;
    ocrBtn.innerText="识别图片文字(OCR)";
    ocrBtn.disabled=false;
}

// 视频预览
const videoFile = document.getElementById('videoFile');
const videoPreview = document.getElementById('videoPreview');
videoFile.onchange = e=>{
    const file = e.target.files[0];
    if(!file)return;
    currentVideoBlob = file;
    const url = URL.createObjectURL(file);
    videoPreview.src = url;
    videoPreview.classList.remove('hidden');
}

// 保存错题
document.getElementById('saveBtn').onclick = ()=>{
    const arr = storage.get();
    const item = {
        id:Date.now(),
        question:document.getElementById('questionText').value,
        reason:document.getElementById('reasonText').value,
        tag:document.getElementById('tagInput').value,
        student:document.getElementById('studentInput').value,
        img:imgPreview.src||"",
        video:currentVideoBlob?URL.createObjectURL(currentVideoBlob):""
    }
    arr.unshift(item);
    storage.save(arr);
    renderList();
    alert("保存成功！");
    //清空表单
    document.getElementById('questionText').value="";
    document.getElementById('reasonText').value="";
    document.getElementById('tagInput').value="";
    document.getElementById('studentInput').value="";
    imgFile.value="";imgPreview.classList.add('hidden');
    videoFile.value="";videoPreview.classList.add('hidden');
    currentVideoBlob=null;
}

//渲染列表
function renderList(){
    const wrap = document.getElementById('listWrap');
    const list = storage.get();
    wrap.innerHTML="";
    list.forEach(item=>{
        const div = document.createElement('div');
        div.className="item-card";
        div.innerHTML=`
            <div><b>学生：</b>${item.student||"未填写"}</div>
            <div><b>知识点：</b><span class="tag">${item.tag||"无标签"}</span></div>
            <div><b>题目：</b><div>${item.question||''}</div></div>
            <div><b>错因：</b>${item.reason||''}</div>
            ${item.img?`<img class="preview-img" src="${item.img}">`:""}
            ${item.video?`<video class="video-preview" controls src="${item.video}"></video>`:""}
            <button onclick="delItem(${item.id})" class="gray">删除</button>
        `
        wrap.appendChild(div);
    })
}
window.delItem = function(id){
    let arr = storage.get();
    arr = arr.filter(x=>x.id!==id);
    storage.save(arr);
    renderList();
}
renderList();
</script>
